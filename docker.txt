services:
  # --- Database MySQL ---
  db:
    image: mysql:8.0
    container_name: oratorio_db
    restart: always
    environment:
      MYSQL_DATABASE: oratorio
      MYSQL_ROOT_PASSWORD: rootpassword
    ports:
      - "3306:3306"
    volumes:
      - mysql_data:/var/lib/mysql
      - ./init_db:/docker-entrypoint-initdb.d

  # --- Flask Backend ---
  backend:
    build: ./backend
    container_name: oratorio_backend
    environment:
      - DB_HOST=db
      - DB_USER=root
      - DB_PASSWORD=rootpassword
      - DB_NAME=oratorio
      - REDIS_HOST=redis
      - REDIS_PORT=6379
    ports:
      - "5000:5000" # Wajib agar React di luar Docker bisa "nembak" ke sini
    depends_on:
      - db
      - redis
    volumes:
      - ./backend/static/uploads:/app/static/uploads

  # --- NGINX REVERSE PROXY ---
  nginx:
    image: nginx:alpine
    container_name: oratorio_proxy
    ports:
      - "8080:80"   # host:8080 → container:80
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf
    depends_on:
      - backend
    # Flutter & React akan akses ke http://localhost:8080 → di-tunnel ngrok
    extra_hosts:
      host.docker.internal: host-gateway

  # --- Redis OTP ----
  redis:
    image: redis:7-alpine
    container_name: oratorio_redis
    restart: always
    ports:
      - "6379:6379"
    volumes:
      - redis_data:/data 

volumes:
  mysql_data:
  redis_data:
